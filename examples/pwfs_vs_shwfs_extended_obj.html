Pyramid vs Shack-Hartmann: extended objects¶

Comparison of the two sensors on extended objects.

Script: example/pwfs_vs_shwfs_extended_obj.py

Result¶

Code¶

"""
Comparison SH-WFS vs P-WFS on extended objects

date: Tue Dec  9 10:20:19 2025
author: rfetick
"""

import numpy as np
import matplotlib.pyplot as plt

from aopera.aopsd import piston_filter
from aopera.turbulence import Atmosphere
from aopera.control import RTC
from aopera.otfpsf import Pupil
from aopera.ffwfs import PWFS
from aopera.shwfs import SHWFS
from aopera.photometry import SourceScience, SourceWFS, Band
from aopera.otfpsf import psd2otf, otf2psf
from aopera.simulation import simulation
from aopera.utils import rad2arcsec

#%% PARAMETERS

### SIMULATION
nx = 700
samp = 3

### SNR
throughput = 0.1
magV = 5
obj_size = np.linspace(0.1, 6, num=10)

### TURBULENCE
seeing = 0.1
windspeed = 0.1

### TELESCOPE, RTC and DM
diameter = 1.5
occultation = 0.25
nact = 17
ao_freq = 1000
frame_delay = 1
ki = 0.5
nmode_ratio = 0.9

### WAVEFRONT SENSORS
r_modul = 3
og_compensation = True
npix_cog = 10
wfs_band = 'R'


#%% INIT CLASSES
src_wvl_nm = Band(wfs_band).wavelength * 1e9

src_sci = SourceScience({'wvl_nm':src_wvl_nm,
                         'zenith_deg':0})

src_wfs = SourceWFS({'wvl_nm':src_wvl_nm,
                     'flux':Band('V').mag2photon(magV) * throughput,
                     'separation':0, 'angle':0, 'size':0})

atm = Atmosphere({'lext':25,
                  'altitude':[0],
                  'wind_speed':[windspeed],
                  'wind_direction':[0],
                  'seeing':seeing,
                  'cn2dh_ratio':[1.0]})

tel = Pupil({'diameter':diameter, 
               'occultation':occultation,
               'nact':nact,
               'nmode_ratio':nmode_ratio})

rtc = RTC({'freq':ao_freq,
           'delay':frame_delay/ao_freq*1000,
           'ki':ki})

emccd = False
ron = 0

shwfs = SHWFS({'lenslet':nact-1,
               'ron':ron,
               'emccd':emccd,
               'samp':1.0,
               'npix_cog':npix_cog})

pwfs = PWFS({'lenslet':nact,
             'ron':ron,
             'emccd':emccd,
             'modulation':r_modul,
             'og_compensation':og_compensation})


#%% AO PSD BUDGET
og = np.zeros(len(obj_size))
var_pwfs = np.zeros(len(obj_size))
var_shwfs = np.zeros(len(obj_size))
strehl_pwfs = np.zeros(len(obj_size))
strehl_shwfs = np.zeros(len(obj_size))

df = tel.frequency_step(samp)
otf_diff = tel.otf_diffraction(nx, samp)
psf_diff = tel.psf_diffraction(nx, samp)

for i in range(len(obj_size)):
    
    print('-'*40)
    print('Iteration %2u/%u'%(i+1,len(obj_size)))
    print()
    
    ### UPDATE OBJECT
    src_wfs.size = obj_size[i]
    pwfs.modulation_sky = (obj_size[i]==0)
    
    ### PWFS SIMULATION
    out = simulation(nx, samp, tel, atm, rtc, pwfs, src_sci, src_wfs, verbose=True)
    psd_total = sum([out.psd[k] for k in out.psd.keys()])
    psf_ao = otf2psf(psd2otf(psd_total, df)*otf_diff)
    og[i] = out.param['og_avg']
    var_pwfs[i] = out.var['ron'] + out.var['photon'] # np.sum(psd_total) * df**2
    strehl_pwfs[i] = 100 * np.max(psf_ao)/np.max(psf_diff)
    
    ### SHWFS SIMULATION
    out = simulation(nx, samp, tel, atm, rtc, shwfs, src_sci, src_wfs, verbose=True)
    psd_total = sum([out.psd[k] for k in out.psd.keys()])
    psf_ao = otf2psf(psd2otf(psd_total, df)*otf_diff)
    var_shwfs[i] = out.var['ron'] + out.var['photon'] # np.sum(psd_total) * df**2
    strehl_shwfs[i] = 100 * np.max(psf_ao)/np.max(psf_diff)
    
    ### PRINT
    print('Strehl P-WFS  : %.1f%%'%strehl_pwfs[i])
    print('Strehl SH-WFS : %.1f%%'%strehl_shwfs[i])
    print()

#%% OPTICAL GAIN GEOMETRIC
def sensi_circle(freq, radius):
    f_r = freq/radius
    low = f_r < 1
    photon_noise = np.sqrt(2)/2
    return photon_noise * (2/np.pi * np.arcsin(f_r*low) * low + 1 * (1-low))


def sensi_disk(freq, radius):
    f_r = freq/radius
    low = f_r < 1
    photon_noise = np.sqrt(2)/2
    return photon_noise * (2/np.pi * ( np.arcsin(f_r*low) + f_r*np.sqrt(1-low*f_r**2) )*low + 1 * (1-low))

xx,yy = (np.mgrid[0:nx,0:nx] - nx//2)*df
ff = np.sqrt(xx**2+yy**2)

def optical_gain_geometric(freq, radius_calib, radius_sky):
    og = sensi_disk(freq, radius_sky) / sensi_circle(freq, radius_calib)
    ao_cutoff = nact/(2*diameter)
    msk = (ff<ao_cutoff)*(ff>df)
    return np.mean(np.sqrt(og[msk]))

obj_diameter_lmbda_D = obj_size / rad2arcsec(src_wvl_nm*1e-9/diameter)
obj_radius = obj_diameter_lmbda_D / 2

og_geo = [optical_gain_geometric(ff, r_modul, obj_radius[i]) for i in range(len(obj_size))]

def psd_pyr(radius_arcsec):
    fcut = pwfs.lenslet/(2*tel.diameter)
    area = (np.abs(fx)<fcut)*(np.abs(fy)<fcut)*piston_filter(ff, tel.diameter) #/ (2*fcut)**2
    pix = tel.diameter / pwfs.lenslet # WFS pixel size in meter in the pupil plane
    pixflt = np.sinc(fx*pix) * np.sinc(fy*pix)
    temp = 2 * rtc.bandwidth_noise/rtc.freq
    FUDGE = np.sqrt(2)
    radius_sky = radius_arcsec / rad2arcsec(src_wvl_nm*1e-9/diameter)
    og = optical_gain_geometric(ff, r_modul, radius_sky)
    p = FUDGE * area / (src_wfs.flux / rtc.freq) / (pixflt*sensi_disk(ff,radius_arcsec)**2 + 1e-8) * temp
    if og_compensation:
        p = p/og**2
    return p

def var_pyr(radius):
    return np.sum(psd_pyr(radius)*tel.frequency_step(samp)**2)

#%% SHACK-HARTMANN SIMPLIFIED PSD
fx,fy = tel.frequency_grid(nx, samp)
ff = np.sqrt(fx**2+fy**2)

def psd_sh(radius_arcsec):
    N_photon_subap = src_wfs.flux / rtc.freq * (tel.diameter/shwfs.lenslet)**2
    dpup = tel.diameter / shwfs.lenslet
    wvl_dpup = rad2arcsec(src_wfs.wvl/dpup)
    wt_wd2 = 1 + 1/shwfs.samp + (2*radius_arcsec/wvl_dpup)**2
    var_subap = np.pi**2/(2*np.log(2))/N_photon_subap * wt_wd2
    pix = tel.diameter / pwfs.lenslet # WFS pixel size in meter in the pupil plane
    pixflt = np.sinc(fx*pix) * np.sinc(fy*pix)
    rec = 1/(2*np.pi*dpup*(pixflt*ff+1e-8))**2
    fcut = shwfs.lenslet/(2*tel.diameter)
    area = (np.abs(fx)<fcut)*(np.abs(fy)<fcut)*piston_filter(ff, tel.diameter) / (2*fcut)**2
    temp = 2 * rtc.bandwidth_noise/rtc.freq
    FUDGE = np.sqrt(2)
    return FUDGE * var_subap * rec * area * temp
    
def var_sh(radius):
    return np.sum(psd_sh(radius)*tel.frequency_step(samp)**2)


#%% PLOT
def setplt():
    plt.grid()
    plt.xlabel('object diameter [$\\lambda/D$]')
    plt.xlim(0, max(obj_radius))

plt.figure(1, figsize=(15,4.5))
plt.clf()

plt.subplot(141)
plt.title('Optical gain')
plt.plot(obj_radius, og, lw=3, label='AOPERA')
plt.plot(obj_radius, og_geo, lw=3, ls=':', c='k', label='geometric')
plt.ylim(0, 1.1)
plt.legend()
setplt()

plt.subplot(142)
plt.title('Variance [rad²] at %unm'%src_wvl_nm)
plt.plot(obj_radius, var_pwfs, lw=3, label='P-WFS')
plt.plot(obj_radius, var_shwfs, lw=3, label='SH-WFS')
plt.plot(obj_radius, var_pwfs[0]*(1+(obj_radius/12)**2), lw=4, ls=':', c='C0')
plt.plot(obj_radius, var_shwfs[0]*(1+2*(obj_radius/shwfs.lenslet)**2), lw=4, ls=':', c='C1')
# plt.plot(obj_radius, [var_pyr(s/2) for s in obj_size], lw=3, ls=':', c='C0', label='P-WFS analytic')
# plt.plot(obj_radius, [var_sh(s/2) for s in obj_size], lw=3, ls=':', c='C1', label='SH-WFS analytic')
plt.ylim(0, 0.7)
plt.legend()
setplt()

plt.subplot(143)
plt.title('Variance ratio SH/P')
plt.plot(obj_radius, var_shwfs/var_pwfs, lw=3)
plt.ylim(0, 6)
setplt()

plt.subplot(144)
plt.title('Strehl [%%] at %unm'%src_wvl_nm)
plt.plot(obj_radius, strehl_pwfs, lw=3, label='P-WFS')
plt.plot(obj_radius, strehl_shwfs, lw=3, label='SH-WFS')
plt.ylim(0, 100)
plt.legend()
setplt()

plt.tight_layout()
