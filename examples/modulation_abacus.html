Modulation abacus¶

Strehl ratio and optical gain against the guide star magnitude, for several pyramid modulations.

Script: example/modulation_abacus.py

Result¶

Code¶

"""
Compute an abacus of optimal modulation for HARMONI-SCAO

date: 2026-04-18
author: rfetick
"""

import numpy as np
import matplotlib.pyplot as plt
from aopera.turbulence import Atmosphere
from aopera.control import RTC
from aopera.otfpsf import Pupil
from aopera.ffwfs import PWFS
from aopera.photometry import SourceScience, SourceWFS, Band
from aopera.otfpsf import psd2otf, otf2psf
from aopera.aopsd import psd_wvl_scaling
from aopera.simulation import simulation

#%% PARAMETERS
file_atmo = 'ohp.ini'
file_instru = 'papyrus.ini'
nx = 512 # number of pixels for simulation
samp = 3 # desired PSF sampling at WFS wavelength
seeing_list = [1.0, 2.0] # atmospheric seeing [arcsec]
wfs_throughput = 0.1 # total photon throughput: atmo, instrument, QE
og_compensation = False
ngs_magV_list = np.linspace(5, 11, num=15)
rmod_list = np.linspace(1, 7, num=3)

#%% READ CONFIG FILE
src_sci = SourceScience.from_file(file_instru)
src_wfs = SourceWFS.from_file(file_instru)
atm = Atmosphere.from_file(file_atmo)
tel = Pupil.from_file(file_instru)
rtc = RTC.from_file(file_instru)
wfs = PWFS.from_file(file_instru)

atm.squeeze_layers(3)
#atm.seeing = seeing # update seeing
wfs.og_compensation = og_compensation

print(src_sci)
print(src_wfs)
print(atm)
print(tel)
print(rtc)
print(wfs)

#%% LOOP on MODULATION and MAGNITUDE
strehl = np.zeros((len(rmod_list),len(ngs_magV_list),len(seeing_list)))
strehl_sci = np.zeros((len(rmod_list),len(ngs_magV_list),len(seeing_list)))
og_avg = np.zeros((len(rmod_list),len(ngs_magV_list),len(seeing_list)))

df = tel.frequency_step(samp)
otf_diff = tel.otf_diffraction(nx, samp)
psf_diff = tel.psf_diffraction(nx, samp)
wvl_wfs_sci = src_wfs.wvl/src_sci.wvl
df_sci = df * wvl_wfs_sci
otf_diff_sci = tel.otf_diffraction(nx, samp/wvl_wfs_sci)
psf_diff_sci = tel.psf_diffraction(nx, samp/wvl_wfs_sci)

for i in range(len(rmod_list)):
    print('\nITERATION %u/%u\n'%(i+1,len(rmod_list)))
    wfs.modulation = rmod_list[i]
    for j in range(len(ngs_magV_list)):
        src_wfs.flux = wfs_throughput * Band('V').mag2photon(ngs_magV_list[j])
        for k in range(len(seeing_list)):
            atm.seeing = seeing_list[k]
            ### PSD at WFS
            result = simulation(nx, samp, tel, atm, rtc, wfs, src_sci, src_wfs, verbose=True, keep_cubes=False, split_fit_temp=True)
            psd_total = sum([result.psd[key] for key in result.psd.keys()])
            psf_ao = otf2psf(psd2otf(psd_total, df)*otf_diff)
            strehl[i,j,k] = 100 * np.max(psf_ao)/np.max(psf_diff)
            og_avg[i,j,k] = result.param['og_avg']
            ### PSD SCALING from WFS WVL to SCIENCE WVL
            psd_total_sci = psd_wvl_scaling(psd_total, wvl_wfs_sci)
            psf_ao_sci = otf2psf(psd2otf(psd_total_sci, df_sci)*otf_diff_sci)
            strehl_sci[i,j,k] = 100 * np.max(psf_ao_sci)/np.max(psf_diff_sci)

#%% PLOT
plt.figure(1)
plt.clf()
for i in range(len(rmod_list)):
    plt.plot(ngs_magV_list, strehl_sci[i,:,1], c='C%u'%i, label='$r_{mod}$=%u $\\lambda/D$'%rmod_list[i])
    plt.plot(ngs_magV_list, strehl_sci[i,:,0], c='C%u'%i, ls='--')
plt.xlabel('Magnitude')
plt.ylabel('Strehl [%%] at %u nm'%src_sci.wvl_nm)
plt.legend()
plt.ylim(0, 80)
plt.xlim(min(ngs_magV_list), max(ngs_magV_list))

plt.tight_layout()


plt.figure(2)
plt.clf()
for i in range(len(rmod_list)):
    plt.plot(ngs_magV_list, og_avg[i,:], c='C%u'%i, label='$r_{mod}$=%u $\\lambda/D$'%rmod_list[i])
plt.xlabel('Magnitude')
plt.ylabel('Optical gain')
plt.legend()
plt.ylim(0,1)
plt.xlim(min(ngs_magV_list), max(ngs_magV_list))
