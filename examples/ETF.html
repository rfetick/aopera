Code¶

"""
Compute ETF in continuous (Laplace) and discrete (Z) transforms, for comparison.
Compute the noise-transfer function (NTF) in continuous transform.
Compare with the analytical formulas for bandwidths.
"""

import numpy as np
import matplotlib.pyplot as plt
from matplotlib.patches import Circle
from scipy.integrate import trapezoid
from aopera.control import open_loop_transfer, closed_loop_transfer, bandwidth, noise_transfer, bandwidth_noise, second_order

Fao = 1000 # AO loop frequency [Hz]
ki = 0.5 # integrator gain
delay_ms = 1.0 # pure delay (WFS frame excluded) [ms] 
db = 0 # choosen cutoff of the ETF [dB]
dbn = -4.5 # chosen cutoff of the NTF [dB]
leak = 0.995

dm_freq = 1000 # DM frequency [Hz]
dm_damp = 0.2 # DM damping

freq = np.logspace(0, np.log10(2*Fao), num=5000)

#%% TRANSFER FUNCTION
frame_delay = delay_ms * 1e-3 * Fao # pure frame delay (WFS integration excluded)

cl_cont = closed_loop_transfer(freq, Fao, ki, frame_delay, discrete=False, leak=leak, dm_freq=dm_freq, dm_damp=dm_damp)
etf2_cont = np.abs(cl_cont)**2
cl_disc = closed_loop_transfer(freq, Fao, ki, frame_delay, discrete=True, leak=leak, dm_freq=dm_freq, dm_damp=dm_damp)
etf2_disc = np.abs(cl_disc)**2

bw = bandwidth(ki, frame_delay, Fao, db=db)

#%% NOISE TRANSFER
ntf2 = np.abs(noise_transfer(freq, Fao, ki, frame_delay, discrete=True, leak=leak, dm_freq=dm_freq, dm_damp=dm_damp))**2

vld = np.where(freq<(Fao/2))
noise_propa = 2*trapezoid(ntf2[vld], freq[vld])/Fao # numerical integration

bwn = bandwidth_noise(ki, frame_delay, Fao, db=dbn)

print('bilateral integral of |NTF|²/Fao : %.2f'%noise_propa)
print('4 * BW / Fao                     : %.2f'%(4*bw/Fao))
print('2 * BW_noise / Fao               : %.2f'%(2*bwn/Fao))

#%% PLOT

def setplt():
    plt.axvline(Fao/2, c='k', ls='--', label='F / 2')
    plt.xlim(min(freq), max(freq))
    plt.grid(visible=True, which='major')
    plt.grid(visible=True, which='minor', alpha=0.3)
    plt.xlabel('Frequency [Hz]')
    plt.legend(loc='lower right')


plt.figure(1, figsize=(17,5))
plt.clf()

plt.subplot(131)
plt.title('DM')
plt.loglog(freq, np.abs(second_order(freq, dm_freq, dm_damp)), lw=2)
plt.ylim(0.1, 10)
setplt()

plt.subplot(132)
plt.title('Transfer function')
plt.loglog(freq, etf2_cont, lw=2, label='ETF continu.')
plt.loglog(freq, etf2_disc, lw=2, label='ETF discrete')
plt.loglog(freq, ntf2, lw=2, label='NTF discrete')
plt.scatter(bw, 10**(db/10), c='C1', label='BW')
plt.axvline(bw, c='C1', ls=':', lw=2)
plt.scatter(bwn, 10**(dbn/10), c='C2', label='BW noise')
plt.axvline(bwn, c='C2', ls=':', lw=2)
plt.ylim(1e-3, 10)
setplt()

oltf = open_loop_transfer(freq, Fao, ki, frame_delay, dm_freq=dm_freq, dm_damp=dm_damp)

plt.subplot(133)
plt.title('Open loop stability criterion')
plt.plot(np.real(oltf), np.imag(oltf), lw=2, label='open loop')
plt.scatter([-1], [0], c='r', label='critical point')
plt.gca().add_artist(Circle((0,0), 1, fc='none', ec='r', ls='--', lw=2, label='|OL|=1'))
plt.plot([-np.sqrt(2)/2,0], [-np.sqrt(2)/2,0], c='r', ls=':', lw=2, label='$\\Delta\\phi_{OL}=45°$')
plt.axis('image')
plt.xticks(np.arange(-2,3))
plt.yticks(np.arange(-2,3))
plt.xlim(-2, 2)
plt.ylim(-2, 2)
plt.xlabel('Real part')
plt.ylabel('Imaginary part')
plt.legend(loc='lower right')
plt.grid()

plt.tight_layout()
