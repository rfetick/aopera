Pyramid vs Shack-Hartmann: noise¶

Comparison of the two sensors.

Script: example/pwfs_vs_shwfs_noise.py

Result¶

Code¶

"""
Comparison between Shack-Hartmann and Pyramid WFS

date: Fri Dec  5 10:10:29 2025
author: rfetick
"""

import numpy as np
import matplotlib.pyplot as plt

from aopera.utils import circavg
from aopera.turbulence import Atmosphere
from aopera.control import RTC
from aopera.otfpsf import Pupil
from aopera.ffwfs import PWFS
from aopera.shwfs import SHWFS
from aopera.photometry import SourceScience, SourceWFS, Band
from aopera.otfpsf import psd2otf, otf2psf
from aopera.simulation import simulation

#%% PARAMETERS

### SIMULATION
nx = 512
samp = 4

### SNR
throughput = 0.1
emccd = False
ron = 0
magV = np.linspace(1, 9, num=12)
obj_size = 2

### TURBULENCE
seeing = 2.5
windspeed = 7

### TELESCOPE, RTC and DM
diameter = 1.5
occultation = 0.25
nact = 35
ao_freq = 1000
frame_delay = 1
ki = 0.5
nmode_ratio = 0.9

### WAVEFRONT SENSORS
r_modul = 3
og_compensation = True
npix_cog = 10
wfs_band = 'R'


#%% INIT CLASSES
src_wvl_nm = Band(wfs_band).wavelength * 1e9

src_sci = SourceScience({'wvl_nm':src_wvl_nm,
                         'zenith_deg':0})

src_wfs = SourceWFS({'wvl_nm':src_wvl_nm,
                     'flux':1e8, # will be updated in loop
                     'separation':0, 'angle':0, 'size':0})

atm = Atmosphere({'lext':25,
                  'altitude':[0],
                  'wind_speed':[windspeed],
                  'wind_direction':[0],
                  'seeing':seeing,
                  'cn2dh_ratio':[1.0]})

tel = Pupil({'diameter':diameter, 
               'occultation':occultation,
               'nact':nact,
               'nmode_ratio':nmode_ratio})

rtc = RTC({'freq':ao_freq,
           'delay':frame_delay/ao_freq*1000,
           'ki':ki})

shwfs = SHWFS({'lenslet':nact-1,
               'ron':ron,
               'emccd':emccd,
               'samp':1.0,
               'npix_cog':npix_cog})

pwfs = PWFS({'lenslet':nact,
             'ron':ron,
             'emccd':emccd,
             'modulation':r_modul})


#%% AO PSD BUDGET

strehl_pwfs = np.zeros(len(magV))
strehl_shwfs = np.zeros(len(magV))

df = tel.frequency_step(samp)
otf_diff = tel.otf_diffraction(nx, samp)
psf_diff = tel.psf_diffraction(nx, samp)

for i in range(len(magV)):
    
    print('-'*40)
    print('Iteration %2u/%u'%(i+1,len(magV)))
    print()
    
    ### UPDATE FLUX and OBJECT
    src_wfs.flux = Band('V').mag2photon(magV[i]) * throughput
    src_wfs.size = obj_size
    pwfs.og_compensation = og_compensation
    pwfs.modulation_sky = (obj_size==0)
    
    ### PWFS SIMULATION
    out = simulation(nx, samp, tel, atm, rtc, pwfs, src_sci, src_wfs, verbose=True)
    psd_total = sum([out.psd[k] for k in out.psd.keys()])
    psf_ao = otf2psf(psd2otf(psd_total, df)*otf_diff)
    strehl_pwfs[i] = 100 * np.max(psf_ao)/np.max(psf_diff)
    
    ### SHWFS SIMULATION
    out = simulation(nx, samp, tel, atm, rtc, shwfs, src_sci, src_wfs, verbose=True)
    psd_total = sum([out.psd[k] for k in out.psd.keys()])
    psf_ao = otf2psf(psd2otf(psd_total, df)*otf_diff)
    strehl_shwfs[i] = 100 * np.max(psf_ao)/np.max(psf_diff)
    
    ### PRINT
    print('Strehl P-WFS  : %.1f%%'%strehl_pwfs[i])
    print('Strehl SH-WFS : %.1f%%'%strehl_shwfs[i])
    print()

#%% PLOT
plt.figure(1)
plt.clf()
plt.title('Strehl [%%] at %unm'%src_wvl_nm)
plt.plot(magV, strehl_pwfs, lw=3, label='P-WFS')
plt.plot(magV, strehl_shwfs, lw=3, label='SH-WFS')
plt.xlabel('mag V')
plt.grid()
plt.xlim(min(magV), max(magV))
plt.yticks(np.arange(0, 110, 5))
plt.ylim(0, 60)
plt.legend()


spot_size = 2
sx,sy = shwfs.sensitivity(*tel.frequency_grid(nx, samp), tel.diameter)
sensi_sh = np.sqrt(np.abs(sx)**2+np.abs(sy)**2)
sensi_pwfs = pwfs.sensitivity(samp, psf_diff, psf_diff)

noise_sh = np.pi**2/(2*np.log(2)) * spot_size**2 / np.clip(sensi_sh**2, 1e-8, np.inf)
noise_pwfs = 1/sensi_pwfs**2

plt.figure(2)
plt.clf()
x = np.arange(nx//2) * tel.frequency_step(samp) * tel.diameter
plt.plot(x, circavg(noise_sh, rmax=nx//2), lw=3, label='SH WFS')
plt.plot(x, circavg(noise_pwfs, rmax=nx//2), lw=3, label='P-WFS')
plt.axvline(pwfs.modulation, c='C1', ls=':', lw=3, label='$r_{mod}$')
plt.xlabel('Frequency [cycle/pup]')
plt.ylabel('Noise propagation')
plt.legend()
plt.ylim(0, 40)
plt.xlim(1, tel.nact/2)
plt.tight_layout()
