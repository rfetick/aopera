Cascaded AO¶

A first-stage result fed into a second AO stage, down to the science PSF.

Script: example/AO_cascaded_simulation.py

Result¶

The script makes 7 figures; the first 6 are shown.

Code¶

"""
Compute phase PSD and PSF for the instrument VLT/SPHERE.

date: Fri Apr 11 11:29:41 2025
author: rfetick
"""

import numpy as np
import matplotlib.pyplot as plt
from matplotlib.colors import LogNorm

from aopera.turbulence import Atmosphere
from aopera.control import RTC
from aopera.otfpsf import Pupil
from aopera.ffwfs import PWFS
from aopera.shwfs import SHWFS
from aopera.photometry import SourceScience, SourceWFS, Band
from aopera.utils import rad2arcsec, circavg
from aopera.otfpsf import psd2otf, otf2psf
from aopera.aopsd import psd_wvl_scaling, psd_cube_wvl_scaling, freq_cube_wvl_scaling
from aopera.simulation import simulation
from aopera.otfpsf import VLTPupil
from aopera.readconfig import read_config_file

#%% PARAMETERS
file_first_stage = 'saxo.ini'
file_second_stage = 'saxoplus.ini'

nx = 512
samp = 2
NB_ITER_OG = 2
IGNORE = ['jitter','ncpa','chromatic','refraction']
CUBE_BINS = 100

#%% READ CONFIG FILE
# Telescope and atmosphere
src_sci = SourceScience.from_file(file_second_stage)
atm = Atmosphere.from_file(file_first_stage)

# First stage
src_wfs = SourceWFS.from_file(file_first_stage)
src_sci_wfs_2 = SourceScience.from_file(file_second_stage)
tel = VLTPupil(read_config_file(file_first_stage, 'pupil'), t_spiders=0.00625, angle_spiders=50.5)
rtc = RTC.from_file(file_first_stage)
wfs = SHWFS.from_file(file_first_stage)

# Second stage
src_wfs_2 = SourceWFS.from_file(file_second_stage)
rtc_2 = RTC.from_file(file_second_stage)
wfs_2 = PWFS.from_file(file_second_stage)
tel_2 = VLTPupil(read_config_file(file_second_stage, 'pupil'), t_spiders=0.00625, angle_spiders=50.5)
samp2 = samp*src_wfs_2.wvl/src_wfs.wvl

print('##### First stage ####')
print(src_sci)
print(src_wfs)
print(atm)
print(tel)
print(rtc)
print(wfs)
print('##### Second stage ####')
print(tel_2)
print(src_wfs_2)
print(rtc_2)
print(wfs_2)



#%% AO first stage
result_first_stage =  simulation(nx, samp, tel, atm, rtc, wfs, src_sci_wfs_2, src_wfs, nu_max=rtc_2.freq, verbose=True, ignore=IGNORE, cube_bins=CUBE_BINS)

psd, var, param = result_first_stage.psd, result_first_stage.var, result_first_stage.param
psd_total = sum([psd[k] for k in psd.keys()])


#%% AO second stage
for k in result_first_stage.psd_cube.keys():
    result_first_stage.psd_cube[k] = psd_cube_wvl_scaling(result_first_stage.psd_cube[k], src_wfs.wvl/src_wfs_2.wvl)
    result_first_stage.frequency_cube[k] = freq_cube_wvl_scaling(result_first_stage.frequency_cube[k], src_wfs.wvl/src_wfs_2.wvl)

result = simulation(nx, samp2, result_first_stage, tel_2, atm, rtc_2, wfs_2, src_sci, src_wfs_2, nu_max=rtc_2.freq/2, verbose=True, nb_iter_og=NB_ITER_OG, ignore=IGNORE, cube_bins=CUBE_BINS)

#%%

psd_total = sum([result.psd[k] for k in result.psd.keys()])

df = tel.frequency_step(samp2)
otf_diff = tel.otf_diffraction(nx, samp2)
psf_diff = tel.psf_diffraction(nx, samp2)
psf_ao = otf2psf(psd2otf(psd_total, df)*otf_diff)
strehl = 100 * np.max(psf_ao)/np.max(psf_diff)

#%% PSD SCALING from WFS WVL to SCIENCE WVL
wvl_wfs_sci = src_wfs_2.wvl / src_sci.wvl 
df_sci = df * wvl_wfs_sci
otf_diff_sci = tel.otf_diffraction(nx, samp2/wvl_wfs_sci)
psd_total_sci = psd_wvl_scaling(psd_total, wvl_wfs_sci)
psf_ao_sci = otf2psf(psd2otf(psd_total_sci, df_sci)*otf_diff_sci)

psf_diff_sci = tel.psf_diffraction(nx, samp2/wvl_wfs_sci)

strehl_sci = 100 * np.max(psf_ao_sci)/np.max(psf_diff_sci)

#%% GET CIRCAVG for PLOT
print()
print('Computing circular averages...')

rad_to_nm = 1e9*src_wfs_2.wvl/(2*np.pi)
psdtotavg = circavg(psd_total) * rad_to_nm**2

psdavg = {}
for k in result.psd.keys():
    psdavg[k] = circavg(result.psd[k]) * rad_to_nm**2

#%% PLOT
lw = 2

plt.figure(1)
plt.clf()

x = np.arange(len(psdtotavg))*df
for k in psdavg.keys():
    plt.loglog(x, psdavg[k], lw=lw, label='%s'%k)
plt.loglog(x, psdtotavg, lw=lw, c='k', ls=':', label='AO total')
plt.xlim(x[1],20)
plt.ylim(1e-1,1e5)
plt.grid()
plt.title('PSD [nm²m²]')
plt.xlabel('Frequency [1/m]')
plt.legend(handlelength=1.2)


def setplt(tab, nc=nx//2, norm=LogNorm(vmin=1e-7,vmax=1e-1), **kwargs):
    pix_size_arcsec = rad2arcsec((src_wfs_2.wvl/tel.diameter) / samp2)
    x = (np.arange(2*nc)-nc) * pix_size_arcsec
    out = plt.pcolormesh(x, x, tab[nx//2-nc:nx//2+nc,nx//2-nc:nx//2+nc], norm=norm, **kwargs)
    plt.axis('image')
    hfov = 0.5
    plt.xlim(-hfov, hfov)
    plt.ylim(-hfov, hfov)
    return out

plt.figure(2, figsize=(8,7))
plt.clf()

plt.subplot(221)
setplt(psf_diff)
plt.ylabel('Position [arcsec]')
plt.title('WFS diffraction')

plt.subplot(222)
setplt(psf_ao)
plt.title('WFS AO : Sr=%.1f%%'%strehl)

plt.subplot(223)
setplt(psf_diff_sci)
plt.ylabel('Position [arcsec]')
plt.title('SCI diffraction')

plt.subplot(224)
out = setplt(psf_ao_sci)
plt.title('SCI AO : Sr=%.1f%%'%strehl_sci)

fig = plt.gcf()
fig.subplots_adjust(right=0.85)
cbar_ax = fig.add_axes([0.89, 0.15, 0.03, 0.7])
fig.colorbar(out, cax=cbar_ax)

plt.show()


psd = result.psd

for l,k in enumerate(psd.keys()):
    plt.figure(l)
    plt.title(list(psd.keys())[l])
    plt.imshow(psd[k]**0.2)
    plt.show()

psd_total_first_stage = sum([result_first_stage.psd[k] for k in result_first_stage.psd.keys()])

plt.figure()
plt.imshow(psd_total_first_stage**0.2)
plt.show(block=False)


plt.figure()
plt.imshow(psd_total**0.2)
plt.show()
