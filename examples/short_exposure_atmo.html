Short exposures without AO¶

Short exposures through the turbulence alone.

Script: example/short_exposure_atmo.py

Code¶

"""
date: Tue Sep 24 11:16:52 2024
author: rfetick
"""

import numpy as np
from numpy.fft import fft2, fftshift
from aopera.utils import aperture
from aopera.turbulence import phase_psd, random_noise, random_sample

import matplotlib.pyplot as plt
import matplotlib.animation as animation

#%% PARAMETERS TO MATCH SIMULATION
nx = 256 # nombre pixel
diameter = 2.0 # [m]
samp = 4.0 # sampling
occultation = 0.2
wind_speed = [2, 3] # [m/s]
wind_direction = [0, 45] # [deg]
cn2dh_ratio = [0.7, 0.3]
nlayer = len(wind_speed) # nb couches atmo
r0 = 0.10 # [m]
lext = 15 # [m]

#%% GENERE UNE PSD DE TEST
df = 1/(samp*diameter) # numerical frequency step [1/m]
fx,fy = (np.mgrid[0:nx,0:nx]-nx//2)*df
freq = np.sqrt(fx**2+fy**2)
psd = phase_psd(freq, r0, lext=lext) #TODO: à remplacer par la PSD corrigée par OA

#%% PARAMETERS FOR SHORT EXP
tmax = 0.1 # [sec]
nt = 100 # number of iterations

#%% APPLICATION
noises = []
for j in range(nlayer):
    noises += [random_noise((nx,nx))]

times = np.linspace(0,tmax,nt)

L = diameter * samp

pup = aperture(nx, samp=samp, occ=occultation)
psf_cube = np.zeros((nt,nx,nx))
nx0 = int(round(nx/samp))
phase_cube = np.zeros((nt,nx0,nx0))

for i in range(len(times)):
    phase = np.zeros((nx,nx))
    for j in range(nlayer):
        t_vxy = times[i]*wind_speed[j]*np.array([np.cos(wind_direction[j]*np.pi/180),np.sin(wind_direction[j]*np.pi/180)])
        phase += random_sample(psd, L, noise=noises[j], t_vxy=t_vxy) * cn2dh_ratio[j]
    
    phase_full = phase * pup
    phase_full[pup==0] = np.nan
    phase_truncated = phase_full[nx//2-nx0//2:nx//2+nx0//2,nx//2-nx0//2:nx//2+nx0//2]
    phase_cube[i,...] = phase_truncated - np.mean(phase[pup>0])
    psf_cube[i,...] = np.abs(fftshift(fft2(fftshift(pup*np.exp(1j*phase)))))**2 / np.sum(pup) / nx**2
    
#%% PLOT
fig = plt.figure(1, figsize=(12,6))
plt.clf()

plt.subplot(121)
plt.title('Phase')
data1 = plt.imshow(phase_cube[0,...])
plt.axis('off')

plt.subplot(122)
plt.title('PSF')
data2 = plt.imshow(psf_cube[0,...])
plt.axis('off')

def update(i):
    fig.suptitle('Time = %3u ms'%(1e3*tmax*i/nt))
    data1.set_data(phase_cube[i,...])
    data2.set_data(psf_cube[i,...])
    return data2


ani = animation.FuncAnimation(fig=fig, func=update, frames=nt, interval=200)
plt.show()
