Pyramid sensitivity¶

RON sensitivity of a pyramid WFS.

Script: example/pwfs_sensitivity.py

Result¶

Code¶

"""
Compute the RON sensitivity of a pyramid WFS.
See figure 4.5 of V.Chambouleyron's PhD thesis.
"""

import numpy as np
import matplotlib.pyplot as plt
from aopera.otfpsf import otf_diffraction, otf2psf
from aopera.ffwfs import ffwfs_sensitivity, modulation, pyramid_mask

#%% PARAMETERS
nx = 512
samp = 8

#%% COMPUTE DIFFRACTION
otf_diff = otf_diffraction(nx, occ=0, samp=samp)
psf_diff = otf2psf(otf_diff)

psf_calib = psf_diff
psf_current = psf_diff

#%% COMPUTE SENSITIVITY
rmod = [0, 1, 3, 5] # modulation radius [lambda/D]
nfaces = [3,4,5] # number of faces

def setplt():
    plt.grid()
    plt.xticks(np.arange(12))
    plt.xlim(0, 12)
    plt.axvline(rmod[i], c='k', ls='--', lw=2)
    plt.xlabel('Frequency [D/$\lambda$]')
    plt.legend(loc='lower right')
    plt.tight_layout()


plt.figure(1, figsize=(15,6))
plt.clf()
plt.suptitle('P-WFS RON sensitivity')
for i in range(len(rmod)):
    plt.subplot(1, len(rmod), i+1)
    plt.title('$R_{mod} = %u \, \\lambda/D$'%rmod[i])
    mod = modulation(nx, rmod[i], samp)
    for j in range(len(nfaces)):
        msk = pyramid_mask(nx, nfaces[j])
        sensi = ffwfs_sensitivity(msk, mod, psf_calib, psf_current)
        sensi_lin = np.diag(sensi)[nx//2:]
        freq = np.arange(len(sensi_lin))/samp*np.sqrt(2)
        plt.plot(freq, sensi_lin, lw=3, label='$N_{face}$ = %u'%nfaces[j])
        plt.axhline(np.sqrt(2)/nfaces[j], c='C%u'%j, ls='--', lw=2)
    plt.yticks(np.arange(1, step=0.05))
    plt.ylim(0, 0.6)
    setplt()


plt.figure(2, figsize=(15,6))
plt.clf()
plt.suptitle('P-WFS photon sensitivity')
for i in range(len(rmod)):
    plt.subplot(1, len(rmod), i+1)
    plt.title('$R_{mod} = %u \, \\lambda/D$'%rmod[i])
    mod = modulation(nx, rmod[i], samp)
    for j in range(len(nfaces)):
        msk = pyramid_mask(nx, nfaces[j])
        sensi = ffwfs_sensitivity(msk, mod, psf_calib, psf_current)*np.sqrt(nfaces[j])
        sensi_lin = np.diag(sensi)[nx//2:]
        freq = np.arange(len(sensi_lin))/samp*np.sqrt(2)
        plt.plot(freq, sensi_lin, lw=3, label='$N_{face}$ = %u'%nfaces[j])
        plt.axhline(np.sqrt(2)/np.sqrt(nfaces[j]), c='C%u'%j, ls='--', lw=2)
    plt.yticks(np.arange(2, step=0.1))
    plt.ylim(0, 1.2)
    setplt()
    
plt.figure(3, figsize=(15,6))
plt.clf()
plt.suptitle('P-WFS ron sensitivity $\\times N_{face}$')
for i in range(len(rmod)):
    plt.subplot(1, len(rmod), i+1)
    plt.title('$R_{mod} = %u \, \\lambda/D$'%rmod[i])
    mod = modulation(nx, rmod[i], samp)
    for j in range(len(nfaces)):
        msk = pyramid_mask(nx, nfaces[j])
        sensi = ffwfs_sensitivity(msk, mod, psf_calib, psf_current)*nfaces[j]
        sensi_lin = np.diag(sensi)[nx//2:]
        freq = np.arange(len(sensi_lin))/samp*np.sqrt(2)
        plt.plot(freq, sensi_lin, lw=3, label='$N_{face}$ = %u'%nfaces[j])
    plt.axhline(np.sqrt(2), c='k', ls='--', lw=2)
    plt.yticks(np.arange(2.2, step=0.2))
    plt.ylim(0, 2)
    setplt()
