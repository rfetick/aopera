Short exposures with AO¶

Minimal AO example without OG computation, with short exposures.

Script: example/short_exposure_ao.py

Result¶

Code¶

"""
AO minimal exemple, without OG computation.
Compute short exposures.

date: 2025-02-16
author: rfetick
"""

import numpy as np
from numpy.fft import fft2, fftshift
import matplotlib.pyplot as plt
from matplotlib.colors import LogNorm
import matplotlib.animation as animation
from matplotlib.patches import Circle

from aopera.turbulence import Atmosphere, random_noise, random_sample
from aopera.control import RTC
from aopera.otfpsf import Pupil
from aopera.ffwfs import PWFS
from aopera.photometry import SourceScience, SourceWFS
from aopera.utils import rad2arcsec, aperture
from aopera.otfpsf import psd2otf, otf2psf
from aopera.fiber import coupling_efficiency
from aopera.simulation import simulation

#%% PARAMETERS
nx = 512
samp = 6

wvl_nm = 1000
zenith_deg = 20
seeing = 2.5
wspd = 7
diameter = 1.5
occ = 0.0
nact = 14 # linear number of actuators across diameter
freq = 500 # loop frequency [Hz]
ki = 0.4 # integrator gain
rmod = 5 # modulation radius [lambda/D]

r_core = 4.5e-6 / 2 # fiber core radius [m]
na = 0.13 # fiber numerical aperture

#%% READ CONFIG FILE
nlayer = 3

src_sci = SourceScience({'wvl_nm':wvl_nm, 'zenith_deg':zenith_deg})
src_wfs = SourceWFS({'wvl_nm':wvl_nm, 'flux':1e8, 'separation':0, 'angle':0, 'size':0})
atm = Atmosphere({'seeing':seeing, 'cn2dh_ratio':nlayer*[1/nlayer,], 'lext':30, 'altitude':nlayer*[0,], 'wind_speed':nlayer*[wspd,], 'wind_direction':np.linspace(0, 90, nlayer)})
tel = Pupil({'diameter':diameter, 'occultation':occ, 'nact':nact})
rtc = RTC({'freq':freq, 'delay':2/freq, 'ki':ki})
wfs = PWFS({'lenslet':nact, 'ron':0, 'emccd':False, 'modulation':rmod})

print(src_sci)
print(src_wfs)
print(atm)
print(tel)
print(rtc)
print(wfs)

#%% GET DIFFRACTION PSF and OTF
otf_diff = tel.otf_diffraction(nx, samp)
psf_diff = tel.psf_diffraction(nx, samp)

#%% AO PSD BUDGET
df = tel.frequency_step(samp) # numerical frequency step [1/m]
fx,fy = (np.mgrid[0:nx,0:nx]-nx//2)*df
freq = np.sqrt(fx**2+fy**2)
psdatmo = atm.phase_psd(freq, src_wfs.wvl, zenith=src_sci.zenith_rad)

result = simulation(nx, samp, tel, atm, rtc, wfs, src_sci, src_wfs, verbose=True)

### COMPUTE PSD TOTAL and PSF
psf_ao = otf2psf(psd2otf(result.psd_total, df)*otf_diff)
strehl_le = 100 * np.max(psf_ao)/np.max(psf_diff)

#%% PLOT
lw = 2

def setplt(tab, nc=100, norm=LogNorm(vmin=1e-6,vmax=np.max(psf_diff)), **kwargs):
    pix_size_arcsec = rad2arcsec((src_wfs.wvl/tel.diameter) / samp)
    x = (np.arange(2*nc)-nc) * pix_size_arcsec
    out = plt.pcolormesh(x, x, tab[nx//2-nc:nx//2+nc,nx//2-nc:nx//2+nc], norm=norm, **kwargs)
    plt.axis('image')
    hfov = 1
    plt.xlim(-hfov, hfov)
    plt.ylim(-hfov, hfov)
    return out

plt.figure(2, figsize=(8,3))
plt.clf()

plt.subplot(121)
setplt(psf_diff)
plt.ylabel('Position [arcsec]')
plt.title('WFS diffraction')

plt.subplot(122)
out = setplt(psf_ao)
plt.title('WFS AO : Sr=%.1f%%'%strehl_le)

fig = plt.gcf()
fig.subplots_adjust(right=0.85)
cbar_ax = fig.add_axes([0.89, 0.15, 0.03, 0.7])
fig.colorbar(out, cax=cbar_ax)

#%% SHORT EXPOSURE
tmax = 0.03 # [sec]
nt = 300 # number of iterations

noises = []
for j in range(atm.nlayer):
    noises += [random_noise((nx,nx))]

times = np.linspace(0,tmax,nt)
L = diameter * samp
pup = aperture(nx, samp=samp, occ=occ)
psf_cube = np.zeros((nt,nx,nx))
phase = np.zeros((nt,nx,nx))
coupling = np.zeros(nt)
strehl = np.zeros(nt)

for i in range(nt):
    print("\rShort exposure %3u / %u"%(i+1,nt), end="")
    for j in range(atm.nlayer):
        t_vxy = times[i]*atm.wind_speed[j]*np.array([np.cos(atm.wind_direction[j]*np.pi/180),np.sin(atm.wind_direction[j]*np.pi/180)])
        psd_layer = result.psd_total * atm.cn2dh_ratio[j]
        phase[i,...] += random_sample(psd_layer, L, noise=noises[j], t_vxy=t_vxy)
    em_field = fftshift(fft2(fftshift(pup*np.exp(1j*phase[i,...]))))
    psf_cube[i,...] = np.abs(em_field)**2 / np.sum(pup) / nx**2
    strehl[i] = 100 * psf_cube[i,nx//2,nx//2] / psf_diff[nx//2,nx//2]
    coupling[i] = 100 * coupling_efficiency(em_field, samp, r_core, na, wvl_nm*1e-9)
    
print()

#%% PLOT
def phi2opd(phi):
    return phi*wvl_nm/(2*np.pi)

fig = plt.figure(3, figsize=(10,3.5))
plt.clf()

ax1 = plt.subplot(131)
plt.title('Phase')
ncopd = nx//(2*samp)
omin = nx//2 - ncopd
omax = nx//2 + ncopd
data1 = plt.imshow(phi2opd(phase[0,omin:omax,omin:omax]), vmin=phi2opd(np.min(phase)), vmax=phi2opd(np.max(phase)))
plt.axis('off')
plt.gca().add_artist(Circle([ncopd,ncopd],nx/samp/2,fc='none',ec='r'))

plt.subplot(132)
plt.title('PSF')
nc = 30
r_core_pix = r_core / (wvl_nm*1e-9/(2*na*samp))
xmin = nx//2-nc
xmax = nx//2+nc
data2 = plt.imshow(psf_cube[0,xmin:xmax,xmin:xmax], vmin=0, vmax=np.max(psf_cube))
plt.axis('off')
plt.gca().add_artist(Circle([nc,nc],r_core_pix,fc='none',ec='r'))

ax3 = plt.subplot(133)
plt.title('Time')
data3, = plt.plot(times, np.zeros(nt), label='coupling')
data4, = plt.plot(times, np.zeros(nt), label='Strehl')
plt.axhline(strehl_le, ls='--', c='C1')
plt.xlim(0, 1e3*tmax)
plt.ylim(0, 80)
plt.xlabel('Time [ms]')
plt.grid()
plt.legend()

plt.tight_layout()

def update(i):
    ax3.set_title('Time = %3u ms'%(1e3*tmax*i/nt))
    ax1.set_title('OPD [%u nm]'%np.std(phi2opd(phase[i][pup])))
    data1.set_data(phi2opd(phase[i,omin:omax,omin:omax]))
    data2.set_data(psf_cube[i,xmin:xmax,xmin:xmax])
    data3.set_data(1e3*times[0:i], coupling[0:i])
    data4.set_data(1e3*times[0:i], strehl[0:i])
    return


ani = animation.FuncAnimation(fig=fig, func=update, frames=nt, interval=100)
plt.show()
