Pyramid pupils¶

Script working with the pyramid signal on a random turbulent phase.

Script: example/pwfs_pupil.py

Result¶

Code¶

"""
date: Sun May  3 18:35:52 2026
author: rfetick
"""

import numpy as np
import matplotlib.pyplot as plt
from aopera.utils import rfftconvolve
from aopera.otfpsf import Pupil
from aopera.ffwfs import PWFS
from aopera.turbulence import Atmosphere, random_sample

#%% PARAMETERS
nx = 512
wvl = 600e-9
samp = 2.2
rmod = 3
seeing = 0.2

#%% COMPUTE
tel = Pupil({'diameter':1, 'occultation':0.3, 'nact':10})
wfs = PWFS({'lenslet':10, 'modulation':rmod, 'ron':0, 'emccd':False})
atm = Atmosphere({'seeing':seeing, 'lext':20, 'altitude':[0], 'cn2dh_ratio':[1], 'wind_speed':[5], 'wind_direction':[0]})

psf = tel.psf_diffraction(nx, samp)
pup = tel.pupil(nx, samp)

fx, fy = tel.frequency_grid(nx, samp)
freq = np.sqrt(fx**2+fy**2)
phase_psd = atm.phase_psd(freq, wvl)
phase = random_sample(phase_psd, tel.diameter*samp)

ir_phase = wfs.impulse_response(samp, psf)
ir_amp = wfs.impulse_response(samp, psf, logamp=True)

calib_image = rfftconvolve(pup, ir_amp)
phase_response = rfftconvolve(phase, ir_phase)

#%% PLOT
plt.figure(1, figsize=(15,5))
plt.clf()

plt.subplot(131)
plt.title('Calibration frame')
plt.imshow(calib_image)
plt.axis('off')

plt.subplot(132)
plt.title('phase')
plt.imshow(phase*pup, cmap='RdBu')
plt.axis('off')

plt.subplot(133)
plt.title('WFS response $\\times$ calib image')
plt.imshow(phase_response*calib_image, cmap='RdBu')
plt.axis('off')
