"""
MMSE reconstructor

date: 2026-05-12
author: rfetick
"""

import numpy as np
import matplotlib.pyplot as plt
from matplotlib.colors import LogNorm

from aopera.turbulence import Atmosphere
from aopera.control import RTC
from aopera.otfpsf import Pupil
from aopera.ffwfs import PWFS
from aopera.photometry import SourceScience, SourceWFS, Band
from aopera.utils import rad2arcsec, circavg
from aopera.simulation import simulation

#%% PARAMETERS
file_atmo = 'ohp.ini'
file_instru = 'ekarus.ini'
nx = 512
samp = 3

seeing = 1.0
zenith_deg = 30
Vmag = 8
ron = 0.2

#%% READ CONFIG FILE
src_sci = SourceScience.from_file(file_instru)
src_wfs = SourceWFS.from_file(file_instru)
atm = Atmosphere.from_file(file_atmo)
tel = Pupil.from_file(file_instru)
rtc = RTC.from_file(file_instru)
wfs = PWFS.from_file(file_instru)

atm.seeing = seeing
src_sci.zenith_deg = zenith_deg
src_sci.wvl_nm = src_wfs.wvl_nm
src_wfs.flux = Band('V').mag2photon(Vmag)
tel.ncpa = 0
wfs.ron = ron
wfs.emccd = False

print(src_sci)
print(src_wfs)
print(atm)
print(tel)
print(rtc)
print(wfs)

#%% PSEUDO-INV RECONSTRUCTOR (DEFAULT)
rtc.set_reconstructor_pinv()

result = simulation(nx, samp, tel, atm, rtc, wfs, src_sci, src_wfs, verbose=True)
psf_ao = result.psf_ao(src_wfs.wvl)
psf_diff = tel.psf_diffraction(nx, samp)
strehl = 100 * np.max(psf_ao)/np.max(psf_diff)

#%% MMSE RECONSTRUCTOR
atm_model = Atmosphere.from_file(file_atmo)
atm_model.seeing = seeing
atm_model.squeeze_layers(1)

rtc.set_reconstructor_mmse(nx, samp, tel, atm_model, wfs, src_sci, src_wfs)

result_mmse = simulation(nx, samp, tel, atm, rtc, wfs, src_sci, src_wfs, verbose=True)
psf_ao_mmse = result_mmse.psf_ao(src_wfs.wvl)
strehl_mmse = 100 * np.max(psf_ao_mmse)/np.max(psf_diff)

#%% GET CIRCAVG for PLOT
print()
print('Computing circular averages...')

df = tel.frequency_step(samp)
rad_to_nm = 1e9*src_wfs.wvl/(2*np.pi)
psdtotavg = circavg(result.psd_total, rmax=nx//2) * rad_to_nm**2
psdtotavg_mmse = circavg(result_mmse.psd_total, rmax=nx//2) * rad_to_nm**2

psdavg = {}
psdavg_mmse = {}
for k in result.psd.keys():
    psdavg[k] = circavg(result.psd[k], rmax=nx//2) * rad_to_nm**2
    psdavg_mmse[k] = circavg(result_mmse.psd[k], rmax=nx//2) * rad_to_nm**2
    
    
#%% PLOT
lw = 2

plt.figure(1, figsize=(10,5))
plt.clf()

x = np.arange(len(psdtotavg))*df

plt.subplot(121)
for k in psdavg.keys():
    plt.loglog(x, psdavg[k], lw=lw, label='%s'%k)
plt.loglog(x, psdtotavg, lw=lw, c='k', ls=':', label='AO total')
plt.xlim(x[1],60)
plt.ylim(1e-2,1e4)
plt.grid()
plt.title('PSD [nm²m²]')
plt.xlabel('Frequency [1/m]')
plt.legend(handlelength=1.2)

plt.subplot(122)
for k in psdavg.keys():
    plt.loglog(x, psdavg_mmse[k], lw=lw, label='%s'%k)
plt.loglog(x, psdtotavg_mmse, lw=lw, c='k', ls=':', label='AO total')
plt.xlim(x[1],60)
plt.ylim(1e-2,1e4)
plt.grid()
plt.title('PSD MMSE [nm²m²]')
plt.xlabel('Frequency [1/m]')
plt.legend(handlelength=1.2)

plt.tight_layout()


def setplt(tab, nc=256, hfov=2, norm=LogNorm(vmin=1e-7,vmax=1e-1), **kwargs):
    pix_size_arcsec = rad2arcsec((src_wfs.wvl/tel.diameter) / samp)
    x = (np.arange(2*nc)-nc) * pix_size_arcsec
    out = plt.pcolormesh(x, x, tab[nx//2-nc:nx//2+nc,nx//2-nc:nx//2+nc], norm=norm, **kwargs)
    plt.axis('image')
    plt.xlim(-hfov, hfov)
    plt.ylim(-hfov, hfov)
    return out

plt.figure(2, figsize=(12,4))
plt.clf()

plt.subplot(131)
setplt(psf_diff)
plt.ylabel('Position [arcsec]')
plt.title('diffraction')

plt.subplot(132)
out = setplt(psf_ao)
plt.title('AO : Sr=%.1f%%'%strehl)

plt.subplot(133)
out = setplt(psf_ao_mmse)
plt.title('AO MMSE : Sr=%.1f%%'%strehl_mmse)

fig = plt.gcf()
fig.subplots_adjust(right=0.85)
cbar_ax = fig.add_axes([0.89, 0.15, 0.03, 0.7])
fig.colorbar(out, cax=cbar_ax)
