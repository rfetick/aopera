"""
Compute phase PSD and PSF for the instrument VLT/SPHERE.

date: Fri Apr 11 11:29:41 2025
author: rfetick
"""

import numpy as np
import matplotlib.pyplot as plt
from matplotlib.colors import LogNorm

from aopera.turbulence import Atmosphere
from aopera.control import RTC
from aopera.otfpsf import Pupil
from aopera.ffwfs import PWFS
from aopera.shwfs import SHWFS
from aopera.photometry import SourceScience, SourceWFS, Band
from aopera.utils import rad2arcsec, circavg
from aopera.otfpsf import psd2otf, otf2psf
from aopera.aopsd import psd_wvl_scaling, psd_cube_wvl_scaling, freq_cube_wvl_scaling
from aopera.simulation import simulation
from aopera.otfpsf import VLTPupil
from aopera.readconfig import read_config_file
from aopera.coronagraph import PerfectCoronagraph

#%% PARAMETERS
file_instru = 'saxo.ini'
CORO_ORDER = 2
nx = 512
samp = 2
IGNORE = ['jitter','ncpa','chromatic','refraction']
CUBE_BINS = 100

#%% READ CONFIG FILE
# Telescope and atmosphere
src_sci = SourceScience.from_file(file_instru)
atm = Atmosphere.from_file(file_instru )

# AO parameters
src_wfs = SourceWFS.from_file(file_instru )
src_sci_wfs_2 = SourceScience.from_file(file_instru )
tel = VLTPupil(read_config_file(file_instru , 'pupil'), t_spiders=0.00625, angle_spiders=50.5)
rtc = RTC.from_file(file_instru )
wfs = SHWFS.from_file(file_instru)

print(src_sci)
print(src_wfs)
print(atm)
print(tel)
print(rtc)
print(wfs)


#%% AO first stage
result_first_stage =  simulation(nx, samp, tel, atm, rtc, wfs, src_sci, src_wfs, verbose=True, ignore=IGNORE, cube_bins=CUBE_BINS)
psd, var, param = result_first_stage.psd, result_first_stage.var, result_first_stage.param
psd_total = sum([psd[k] for k in psd.keys()])

#%% Coronagraph
wvl_wfs_sci = src_wfs.wvl / src_sci.wvl 
samp_sci = samp / wvl_wfs_sci
psd_total_sci = psd_wvl_scaling(psd_total, wvl_wfs_sci)
coro = PerfectCoronagraph(tel, nx, samp_sci, order=CORO_ORDER)
contrast_map = coro.raw_contrast(psd_total_sci)
sep, contrast = coro.raw_contrast_curve(psd_total_sci)

#%% Plot
hfov = 30 # half field of view of the plots [lambda/D]
ext = nx / (2*samp_sci)
norm = LogNorm(vmin=1e-7, vmax=1e-2)

plt.figure(1, figsize=(10,4.2))
plt.clf()

plt.subplot(121)
plt.imshow(contrast_map, extent=(-ext,ext)*2, origin='lower', norm=norm)
plt.colorbar(label='Raw contrast')
plt.xlim(-hfov, hfov)
plt.ylim(-hfov, hfov)
plt.xlabel(r'Position [$\lambda/D$]')
plt.ylabel(r'Position [$\lambda/D$]')
plt.title('Raw contrast map')

plt.subplot(122)
plt.semilogy(sep, contrast, lw=2)
plt.xlim(0, hfov)
plt.ylim(norm.vmin, norm.vmax)
plt.grid(which='both', alpha=0.3)
plt.xlabel(r'Separation [$\lambda/D$]')
plt.ylabel('Azimuthal average of the raw contrast')
plt.title('Raw contrast curve')

plt.tight_layout()
plt.show()
