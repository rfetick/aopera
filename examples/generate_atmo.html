Generate an atmosphere¶

Random phase and log-amplitude screens from a turbulence profile.

Script: example/generate_atmo.py

Result¶

Code¶

"""
Generate an atmosphere
"""

import numpy as np
import matplotlib.pyplot as plt
from aopera.turbulence import cn2dh_to_r0, phase_psd, random_sample, logamp_psd, logamp_variance

#%% PARAMETERS

### SOURCE
wvl = 500e-9
zenith = 30  # zenith angle [deg]
src_alt = np.inf

### ATMOSPHERE
cn2dh = np.array([1, 0.2])*1e-12  # layers cn2dh
alt = [0, 10000]  # layers altitude [m]
lext = 15  # Von-Karman outer scale [m]
lint = 5e-3  # Von-Karman internal scale [m]

### SIMULATION
npix = 1024  # number of pixels
length = 2  # size of phase screen array [m]

#%% COMPUTE r0 and log-amplitude variance
zenith_rad = zenith*np.pi/180
r0 = cn2dh_to_r0(cn2dh, wvl, zenith=zenith_rad)
logamp_var = logamp_variance(alt, cn2dh, wvl, zenith=zenith_rad)
print("r0 : %.1f cm (wvl=%unm, z=%u°)"%(r0*100, wvl*1e9,zenith))
print("log-amp variance : %.2f"%logamp_var)

#%% GET PSD 1D
f_1d = np.logspace(-2, 3, num=npix)
psd_1d = phase_psd(f_1d, r0, lext=lext, lint=lint)
psd_1d_logamp = logamp_psd(f_1d, alt, cn2dh, wvl, lext=lext, lint=lint, zenith=zenith_rad)

#%% GET PSD 2D
fx,fy = (np.mgrid[0:npix, 0:npix] - npix/2)/length
f_2d = np.sqrt(fx**2 + fy**2)
psd_2d = phase_psd(f_2d, r0, lext=lext, lint=lint)
psd_2d_logamp = logamp_psd(f_2d, alt, cn2dh, wvl, lext=lext, lint=lint, zenith=zenith_rad)

#%% PLOT
plt.figure(1, figsize=(10,5))
plt.clf()

plt.subplot(121)
plt.title("OPD PSD [um² m²]")
plt.loglog(f_1d, psd_1d*(wvl*1e6/(2*np.pi))**2, lw=2)
plt.xlim(1e-2, 1e3)
plt.ylim(1e-14, 1e4)
plt.grid()
plt.xlabel("Frequency [1/m]")

plt.subplot(122)
plt.title("Log-amplitude PSD [m²]")
plt.loglog(f_1d, psd_1d_logamp, lw=2)
plt.xlim(1e-2, 1e3)
plt.ylim(1e-8, 1e-2)
plt.grid()
plt.xlabel("Frequency [1/m]")



x = np.linspace(0, length, num=npix+1)

plt.figure(2, figsize=(10,5))
plt.clf()

plt.subplot(121)
plt.title("OPD [um]")
plt.pcolormesh(x, x, random_sample(psd_2d, length)*(wvl*1e6/(2*np.pi)))
plt.axis('image')
plt.colorbar()
plt.xlabel("X [m]")
plt.ylabel("Y [m]")

plt.subplot(122)
plt.title("Log-amplitude")
plt.pcolormesh(x, x, random_sample(psd_2d_logamp, length))
plt.axis('image')
plt.colorbar()
plt.xlabel("X [m]")
plt.ylabel("Y [m]")
