Pyramid log-amplitude sensitivity¶

Phase and log-amplitude sensitivity of a pyramid WFS.

Script: example/pwfs_logamp_sensitivity.py

Result¶

Code¶

"""
Compute P-WFS phase sensitivity and log-amplitude sensitivity

date: Fri Apr 10 11:50:08 2026
author: rfetick
"""

import numpy as np
from numpy.fft import fft2, fftshift, ifft2, ifftshift
import matplotlib.pyplot as plt
from aopera.otfpsf import otf_diffraction, otf2psf
from aopera.ffwfs import modulation, pyramid_mask, ffwfs_transfer_function

#%% PARAMETERS
nx = 512
samp = 8
rmod = 3

#%% COMPUTE
def fftconvolve(a, b):
    return np.real(ifftshift(ifft2(fft2(fftshift(a))*fft2(fftshift(b)))))

otf_diff = otf_diffraction(nx, occ=0, samp=samp)
psf_diff = otf2psf(otf_diff)

mask = pyramid_mask(nx, 4)
modu_pth = modulation(nx, rmod, samp)

tf_phi = ffwfs_transfer_function(mask, modu_pth, psf_diff, logamp=False)
tf_chi = ffwfs_transfer_function(mask, modu_pth, psf_diff, logamp=True)

sensi_phi = np.sqrt(fftconvolve(np.abs(tf_phi)**2, psf_diff))
sensi_chi = np.sqrt(fftconvolve(np.abs(tf_chi)**2, psf_diff))
sensi_cross = np.sqrt(np.abs(fftconvolve(tf_chi*np.conjugate(tf_phi), psf_diff)))

rec = np.conjugate(tf_phi)/np.abs(tf_phi)**2

#%% PLOT
plt.figure(1)
plt.clf()
plt.subplot(131)
plt.title('Phase sensitivity')
plt.imshow(sensi_phi, vmin=0, cmap='Spectral_r')
plt.colorbar()
plt.subplot(132)
plt.title('Log-amplitude sensitivity')
plt.imshow(sensi_chi, vmin=0, cmap='Spectral_r')
plt.colorbar()
plt.subplot(133)
plt.title('Cross-sensitivity')
plt.imshow(sensi_cross, vmin=0, cmap='Spectral_r')
plt.colorbar()

plt.figure(2)
plt.clf()
plt.title('Sensitivity')
y = np.diag(sensi_phi[nx//2:,nx//2:])
x = np.arange(len(y)) / samp *np.sqrt(2)
plt.plot(x, y)
plt.axvline(rmod, c='k', ls=':')
plt.xlabel('Frequency [cycle/pup]')
plt.xlim(0, 10)
plt.ylim(0, 0.5)
