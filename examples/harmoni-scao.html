"""
Compute phase PSD and PSF for the instrument ELT/HARMONI-SCAO.

date: 2025-04-29
author: rfetick
"""

import numpy as np
import matplotlib.pyplot as plt
from matplotlib.colors import LogNorm

from aopera.turbulence import Atmosphere
from aopera.control import RTC
from aopera.otfpsf import Pupil
from aopera.ffwfs import PWFS
from aopera.photometry import SourceScience, SourceWFS, Band
from aopera.utils import rad2arcsec
from aopera.otfpsf import psd2otf, otf2psf
from aopera.aopsd import psd_wvl_scaling
from aopera.simulation import simulation

#%% PARAMETERS
file_atmo = 'paranal.ini'
file_instru = 'harmoni-scao.ini'
nx = 1200 # number of pixels for simulation
samp = 3 # desired PSF sampling at WFS wavelength
seeing = 0.8 # atmospheric seeing [arcsec]
ngs_magV = 11

#%% READ CONFIG FILE
src_sci = SourceScience.from_file(file_instru)
src_wfs = SourceWFS.from_file(file_instru)
atm = Atmosphere.from_file(file_atmo)
tel = Pupil.from_file(file_instru)
rtc = RTC.from_file(file_instru)
wfs = PWFS.from_file(file_instru)

atm.squeeze_layers(3)
atm.seeing = seeing # update seeing
src_wfs.flux = Band('V').mag2photon(ngs_magV)

diffraction_arcsec = rad2arcsec(src_wfs.wvl/tel.diameter)
modul_arcsec = wfs.modulation * diffraction_arcsec
wfs.modulation_sky = (src_wfs.size<modul_arcsec) # modulate or no on-sky

print(src_sci)
print(src_wfs)
print(atm)
print(tel)
print(rtc)
print(wfs)

print()
if wfs.modulation_sky:
    print('Sky modulation activated')
else:
    print('Sky modulation UNACTIVATED')
print()

#%% AO PSD BUDGET
result = simulation(nx, samp, tel, atm, rtc, wfs, src_sci, src_wfs, verbose=True)
psd_total = sum([result.psd[k] for k in result.psd.keys()])

df = tel.frequency_step(samp)
otf_diff = tel.otf_diffraction(nx, samp)
psf_diff = tel.psf_diffraction(nx, samp)
psf_ao = otf2psf(psd2otf(psd_total, df)*otf_diff)
strehl = 100 * np.max(psf_ao)/np.max(psf_diff)

#%% PSD SCALING from WFS WVL to SCIENCE WVL
wvl_wfs_sci = src_wfs.wvl/src_sci.wvl
df_sci = df * wvl_wfs_sci
otf_diff_sci = tel.otf_diffraction(nx, samp/wvl_wfs_sci)
psd_total_sci = psd_wvl_scaling(psd_total, wvl_wfs_sci)
psf_ao_sci = otf2psf(psd2otf(psd_total_sci, df_sci)*otf_diff_sci)

psf_diff_sci = tel.psf_diffraction(nx, samp/wvl_wfs_sci)

strehl_sci = 100 * np.max(psf_ao_sci)/np.max(psf_diff_sci)

#%% TURBULENT PSF
fx,fy = np.mgrid[0:nx,0:nx]-nx//2
freq = np.sqrt(fx**2+fy**2)*df
psd_turb_wfs = atm.phase_psd(freq, src_wfs.wvl, zenith=src_sci.zenith_rad)
psd_turb_sci = psd_wvl_scaling(psd_turb_wfs, wvl_wfs_sci)

psf_turb_wfs = otf2psf(psd2otf(psd_turb_wfs, df)*otf_diff)
psf_turb_sci = otf2psf(psd2otf(psd_turb_sci, df_sci)*otf_diff_sci)

#%% GET CIRCAVG for PLOT
print()
print('Computing circular averages...')

def circavg(tab):
    """Fast computation of avg on X and Y"""
    nx = tab.shape[0]
    return 0.5*(tab[nx//2,nx//2:]+tab[nx//2:,nx//2])

rad_to_nm = 1e9*src_wfs.wvl/(2*np.pi)
psdtotavg = circavg(psd_total) * rad_to_nm**2

psdavg = {}
for k in result.psd.keys():
    psdavg[k] = circavg(result.psd[k]) * rad_to_nm**2

#%% PLOT
lw = 2

plt.figure(1, figsize=(8,4))
plt.clf()
x = np.arange(len(psdtotavg))*df

plt.subplot(121)
for k in psdavg.keys():
    plt.loglog(x, psdavg[k], lw=lw, label='%s'%k)
plt.loglog(x, psdtotavg, lw=lw, c='k', ls=':', label='AO total')
plt.xlim(x[1], 10)
plt.ylim(1, 1e6)
plt.grid()
plt.title('PSD [nm²m²]')
plt.xlabel('Frequency [1/m]')
# plt.legend(handlelength=1.2)

plt.subplot(122)
for k in psdavg.keys():
    plt.semilogx(x, np.sqrt(2*np.pi*np.cumsum(psdavg[k]*x*df)), lw=lw, label='%s'%k)
plt.semilogx(x, np.sqrt(2*np.pi*np.cumsum(psdtotavg*x*df)), lw=lw, c='k', ls=':', label='AO total')
plt.xlim(x[1], 10)
plt.ylim(0, 200)
plt.grid()
plt.title('$\\sigma(f)=\\sqrt{2\\pi\\int_0^f PSD(u)udu}$ [nm]')
plt.xlabel('Frequency [1/m]')
plt.legend(handlelength=1.2)

plt.tight_layout()


def setplt(tab, nc=512, norm=LogNorm(vmin=1e-7,vmax=1e-1), **kwargs):
    pix_size_arcsec = rad2arcsec((src_wfs.wvl/tel.diameter) / samp)
    x = (np.arange(2*nc)-nc) * pix_size_arcsec * 1e3
    out = plt.pcolormesh(x, x, tab[nx//2-nc:nx//2+nc,nx//2-nc:nx//2+nc], norm=norm, **kwargs)
    plt.axis('image')
    hfov = 500
    plt.xlim(-hfov, hfov)
    plt.ylim(-hfov, hfov)
    return out

plt.figure(2, figsize=(14,9))
plt.clf()
plt.suptitle('PSF at %u nm (top) and %u nm (bottom)'%(src_wfs.wvl_nm, src_sci.wvl_nm))

plt.subplot(231)
setplt(psf_diff)
plt.ylabel('Position [mas]')
plt.title('Diffraction')

plt.subplot(232)
setplt(psf_ao)
plt.title('AO : Sr=%.1f%%'%strehl)

plt.subplot(233)
setplt(psf_turb_wfs)
plt.title('Seeing limited')

plt.subplot(234)
setplt(psf_diff_sci)
plt.ylabel('Position [mas]')
plt.title('Diffraction')

plt.subplot(235)
out = setplt(psf_ao_sci)
plt.title('AO : Sr=%.1f%%'%strehl_sci)

plt.subplot(236)
setplt(psf_turb_sci)
plt.title('Seeing limited')

fig = plt.gcf()
fig.subplots_adjust(right=0.85)
cbar_ax = fig.add_axes([0.89, 0.15, 0.03, 0.7])
fig.colorbar(out, cax=cbar_ax)
