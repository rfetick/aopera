Tip-tilt temporal PSD¶

Temporal PSD of the turbulence tip-tilt.

Script: example/turbulence_tiptilt.py

Result¶

Code¶

"""
Compute turbulence tip-tilt temporal PSD

date: Thu Dec 11 13:37:38 2025
author: rfetick
"""

import numpy as np
import matplotlib.pyplot as plt
from aopera.zernike import zernike_fourier
from aopera.turbulence import Atmosphere
from aopera.utils import rad2arcsec
from aopera.control import closed_loop_transfer

#%% PARAMETERS
dft = 0.1 # temporal frequency step [Hz]
fmax = 100 # maximal temporal frequency

diameter = 1.5 # [m]
seeing = 2.0 # [arcsec]
L0 = 15 # external scale [m]
windspeed_eq = 10 # equivalent wind speed [m/s]

#%% Load atmosphere
atm = Atmosphere.from_file('ohp.ini')
atm.seeing = seeing
atm.wind_speed *= windspeed_eq / atm.equivalent_wind_speed
atm.squeeze_layers(10)
print(atm)

#%% COMPUTE TT PSD

npix = 2*round(fmax/dft)

temp_psd_tip = np.zeros(npix)
temp_psd_tilt = np.zeros(npix)

for i in range(atm.nlayer):
    
    print('Iteration %2u/%u'%(i+1,atm.nlayer))
    
    windspeed = atm.wind_speed[i]
    cn2dh_r = atm.cn2dh_ratio[i]
    
    dfx = dft/windspeed
    dx = 1/(dfx*npix)
    samp = dx*npix/diameter
    
    tt = zernike_fourier(1, -1, npix, samp=samp, norm='ansi')
    tt_psd = np.abs(tt)**2
    
    tt_psd *= 1 / (np.sum(tt_psd)*dfx**2)
    fx,fy = (np.mgrid[0:npix,0:npix]-npix//2) * dfx
    vonk = 1e12 * 0.0097 * atm.cn2dh[i] * (fx**2 + fy**2 + 1/L0**2)**(-11/6)
    temp_psd_tip += 1/windspeed * np.sum(tt_psd * vonk, axis=0) * dfx
    temp_psd_tilt += 1/windspeed * np.sum(tt_psd * vonk, axis=1) * dfx
    

#%% PLOT
ft = (np.arange(npix)-npix//2)*dft
ft_mono = ft[npix//2:]

ao_freq = 2000
ki = 0.5
frame_delay = 1
lqg_factor = 0.25
cl = closed_loop_transfer(ft_mono, ao_freq, ki, frame_delay)
etf = lqg_factor * np.abs(cl)**2
etf[0] = 0 # avoid issue at f=0

temp_psd_tip_mono = 2 * temp_psd_tip[npix//2:]
temp_psd_tilt_mono = 2 * temp_psd_tilt[npix//2:]
temp_psd_mono = temp_psd_tip_mono + temp_psd_tilt_mono

um2mas = rad2arcsec(4 / diameter) * 1e-6 * 1e3

def setplt():
    plt.xlim(dft, fmax)
    plt.grid()
    plt.axvline(0.3*2*atm.equivalent_wind_speed/diameter, c='k', ls=':', label='$0.6\\, V/D$')
    plt.legend()

plt.figure(1, figsize=(10,8))
plt.clf()
plt.suptitle('Input (top) and after AO (bottom)')

for i in range(2):
    
    e = [1,etf][i]
    
    plt.subplot(2,2,2*i+1)
    plt.title('PSD [mas²/Hz]')
    plt.loglog(ft_mono, e * um2mas**2 * temp_psd_tip_mono, lw=1, label='tip')
    plt.loglog(ft_mono, e * um2mas**2 * temp_psd_tilt_mono, lw=1, label='tilt')
    plt.loglog(ft_mono, e * um2mas**2 * temp_psd_mono, lw=2, label='sum', c='k')
    plt.ylim(1e-3, 1e6)
    plt.xlabel('Temporal frequency [Hz]')
    setplt()
    
    plt.subplot(2,2,2*i+2)
    plt.title('$\\sigma(f) = \\sqrt{\\int_0^f PSD(u)du}$ [mas]')
    plt.semilogx(ft_mono, um2mas * np.sqrt(np.cumsum(e*temp_psd_tip_mono)*dft), lw=1, label='tip')
    plt.semilogx(ft_mono, um2mas * np.sqrt(np.cumsum(e*temp_psd_tilt_mono)*dft), lw=1, label='tilt')
    plt.semilogx(ft_mono, um2mas * np.sqrt(np.cumsum(e*temp_psd_mono)*dft), lw=2, label='sum', c='k')
    plt.ylim(bottom=0)
    plt.xlabel('Temporal frequency [Hz]')
    setplt()

plt.tight_layout()
